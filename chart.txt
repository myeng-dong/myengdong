Sub CreateDynamicOrgChart_RatioAdjusted()
    Dim ws As Worksheet
    Set ws = ActiveSheet
    
    ' 1. 배열 형태로 하위 계층 수 입력 받기
    Dim inputStr As String
    inputStr = InputBox("각 상위계층(헤더)이 가질 하위계층(자식)의 수를 쉼표로 구분하여 입력하세요." & vbCrLf & _
                        "(예: 3, 2, 3)", "조직도 배열 입력", "3, 2, 3")
    
    If Trim(inputStr) = "" Then Exit Sub
    
    Dim arrChildren() As String
    arrChildren = Split(inputStr, ",")
    
    Dim numHeaders As Integer
    numHeaders = UBound(arrChildren) - LBound(arrChildren) + 1
    
    ' 화면 업데이트 중지
    Application.ScreenUpdating = False
    
    ' 기존 시트 내용 및 서식 초기화
    ws.Cells.Clear
    ws.Cells.Borders.LineStyle = xlNone
    ws.Cells.Interior.ColorIndex = xlNone
    
    ' --- 기본 행 높이 설정 ---
    ws.Cells.RowHeight = 24
    
    ' --- 데이터 열과 간격 열의 너비 변수 설정 (30% 비율) ---
    Dim dataWidth As Double: dataWidth = 8  ' 박스 너비 크게 확대
    Dim gapWidth As Double: gapWidth = 2     ' 박스 너비의 30% 수준으로 간격 축소
    
    Dim h As Integer, i As Integer, r As Integer
    Dim currentGroupStartCol As Integer
    currentGroupStartCol = 2 ' B열부터 시작
    
    ' --- 전체 그리드(열 너비) 사전 계산 및 일괄 설정 ---
    Dim totalCols As Integer
    totalCols = 2
    For h = 0 To numHeaders - 1
        totalCols = totalCols + CInt(Trim(arrChildren(h))) * 4
    Next h
    
    Dim c As Integer
    For c = 1 To totalCols + 2
        ' (c-2)를 4로 나눈 나머지가 0, 1이면 데이터 열, 2, 3이면 간격 열
        If (c - 2) Mod 4 < 2 And c >= 2 Then
            ws.Columns(c).ColumnWidth = dataWidth  ' 데이터 열 배정 (너비 10)
        Else
            ws.Columns(c).ColumnWidth = gapWidth   ' 간격 열 배정 (너비 3)
        End If
    Next c
    
    ' 헤더 그룹만큼 반복
    For h = 0 To numHeaders - 1
        Dim numChildren As Integer
        On Error Resume Next
        numChildren = CInt(Trim(arrChildren(h)))
        On Error GoTo 0
        
        If numChildren > 0 Then
            Dim startCol As Integer
            startCol = currentGroupStartCol
            
            ' 상위 계층의 정중앙 테두리 위치 계산 (자식 배열의 정중앙)
            Dim mainLineCol As Integer
            mainLineCol = startCol + (numChildren - 1) * 2
            
            ' --- 상위계층 크기 고정 (센터를 기준으로 항상 4칸 배정) ---
            Dim topLeftCol As Integer
            Dim topRightCol As Integer
            topLeftCol = mainLineCol - 1
            topRightCol = mainLineCol + 2
            
            ' 상위 계층 박스 그리기 (3~6행)
            For r = 3 To 6
                With ws.Range(ws.Cells(r, topLeftCol), ws.Cells(r, topRightCol))
                    .Borders.Weight = xlThin
                End With
            Next r
            ws.Range(ws.Cells(3, topLeftCol), ws.Cells(6, topRightCol)).BorderAround Weight:=xlMedium
            
            ' --- 상위계층에서 잇는 메인 줄은 7, 8행 2칸만 ---
            ws.Range(ws.Cells(7, mainLineCol), ws.Cells(8, mainLineCol)).Borders(xlEdgeRight).Weight = xlThin
            
            ' 가로 연결선 (8행 하단)
            If numChildren > 1 Then
                Dim horizEndCol As Integer
                horizEndCol = startCol + (numChildren - 1) * 4
                ws.Range(ws.Cells(8, startCol + 1), ws.Cells(8, horizEndCol)).Borders(xlEdgeBottom).Weight = xlThin
            End If
            
            ' --- 하위 계층 개별 선 및 박스 그리기 ---
            For i = 1 To numChildren
                Dim childCol As Integer
                childCol = startCol + (i - 1) * 4
                
                ' 하위계층으로 떨어지는 개별 세로선 (9행)
                ws.Cells(9, childCol).Borders(xlEdgeRight).Weight = xlThin
                
                ' 하위 박스 그리기 (10행 ~ 22행) - 내부 격자 2열 구조
                With ws.Range(ws.Cells(10, childCol), ws.Cells(22, childCol + 1))
                    .Borders.Weight = xlThin
                    .BorderAround Weight:=xlMedium
                End With
            Next i
            
            ' 다음 상위계층(그룹) 생성을 위해 시작 열 위치 이동
            currentGroupStartCol = startCol + (numChildren * 4)
        End If
    Next h
    
    ' --- A4 가로(Landscape) 인쇄 영역 딱 맞춤 설정 ---
    With ws.PageSetup
        .PaperSize = xlPaperA4
        .Orientation = xlLandscape
        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = 1
        .CenterHorizontally = True
        .CenterVertically = False
        .LeftMargin = Application.InchesToPoints(0.2)
        .RightMargin = Application.InchesToPoints(0.2)
        .TopMargin = Application.InchesToPoints(0.5)
        .BottomMargin = Application.InchesToPoints(0.5)
    End With
    
    Application.ScreenUpdating = True
    MsgBox "데이터 박스는 넓어지고 간격은 30%로 축소된 조직도가 완성되었습니다!", vbInformation
End Sub
